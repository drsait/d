2. Android Paylaş Menüsü Kodu (MainActivity.kt)

Ekteki video_donusturucu_android_proje.zip dosyasında yer alan bu Android kodu, YouTube/Instagram/Facebook'ta Paylaş butonuna bastığında linki yakalayıp otomatik dönüştürür: 

package com.videodonusturucu.app

import android.app.Activity
import android.content.Intent
import android.net.Uri
import android.os.Bundle

class MainActivity : Activity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val rawText = intent?.getStringExtra(Intent.EXTRA_TEXT)
            ?: intent?.dataString
        if (!rawText.isNullOrBlank()) {
            val extracted = extractUrl(rawText)
            val targetUrl = convertUrl(extracted)
            startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(targetUrl)))
            finish()
        }
    }

    private fun extractUrl(text: String): String {
        val regex = Regex("https?://[^\\s]+")
        return regex.find(text)?.value ?: text.trim()
    }

    private fun convertUrl(url: String): String {
        val host = Uri.parse(url).host?.lowercase() ?: ""
        return when {
            host.contains("youtube.com") -> url.replace("youtube.com", "vidsyoutube.com")
            host.contains("instagram.com") -> "https://saveclip.app/en8?q=$url"
            host.contains("facebook.com") -> "https://fsaver.com/en?q=$url"
            else -> "https://vidssave.com/watch?l=$url"
        }
    }
}
